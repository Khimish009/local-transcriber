# ML dependencies — installed only into the worker image, never into the API image.
#
# The torch stack is pinned to `+cpu` local versions, which exist ONLY on
# https://download.pytorch.org/whl/cpu (see backend/Dockerfile). This is what keeps the
# base compose a portable CPU build: the plain PyPI wheels bundle CUDA libraries
# (cuDNN alone is ~650 MB) on both amd64 and arm64.
#
# Versions must stay mutually compatible: torchcodec 0.16 requires torch >= 2.11, and
# torchaudio 2.11 is the last release of that line. Bump all three together.
torch==2.11.0+cpu
torchaudio==2.11.0+cpu
torchcodec==0.16.0+cpu

# Exact pin: worker/pipeline/embeddings.py replaces SpeakerDiarization.get_embeddings, an
# internal of this version (TASKS.md T12.1). After a bump, run `python -m
# benchmark.equivalence` — it compares the result with stock pyannote.
pyannote.audio==4.0.7

# GigaAM ASR. PyPI still carries 0.1.0, which predates the v3 models and word-level
# timestamps and pins torch<=2.5.1 — so the package is installed from the repository at a
# fixed commit. torch is an optional extra there, which is why it does not fight the
# `+cpu` pins above. worker/pipeline/asr.py batches through the private `_decode` of this
# commit (TASKS.md T12.2): after a bump, run `python -m benchmark.equivalence --only asr`.
gigaam @ git+https://github.com/salute-developers/GigaAM.git@7447938d791c4f3e643386ee22c33777004293a5
